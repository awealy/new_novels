async submitForm() { const title = document.getElementById('novelTitle').value.trim(); const author = document.getElementById('novelAuthor').value.trim(); const desc = document.getElementById('novelDescription').value.trim(); const tags = document.getElementById('novelTags').value.trim(); const files = this.files; const token = "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"; // 临时调试用 if (!title || !author || !desc || !tags || files.length === 0) { this.showStatus('请填写完整信息并上传至少一个文件', 'error'); return; } const owner = "awealy"; const repo = "new_novels"; const branch = "main"; const folder = `uploads/pending/${Date.now()}_${title.replace(/\s+/g, "_")}`; const metadata = { title, author, description: desc, tags: tags.split(',').map(t => t.trim()), uploadTime: new Date().toISOString() }; // 上传 metadata.json await uploadFile(`${folder}/metadata.json`, JSON.stringify(metadata, null, 2)); // 上传所有章节 for (const file of files) { const content = await file.text(); await uploadFile(`${folder}/${file.name}`, content); } this.showStatus('✅ 小说上传成功！GitHub Actions 将自动处理。', 'success'); async function uploadFile(path, content) { const encoded = btoa(unescape(encodeURIComponent(content))); const response = await fetch(`https://api.github.com/repos/${owner}/${repo}/contents/${path}`, { method: "PUT", headers: { "Authorization": `Bearer ${token}`, "Content-Type": "application/json" }, body: JSON.stringify({ message: `上传小说文件: ${path}`, content: encoded, branch }) }); if (!response.ok) { const data = await response.json(); throw new Error(data.message); } } }
